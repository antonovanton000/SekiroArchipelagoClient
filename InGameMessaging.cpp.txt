// =============================================================
// Константы для наших кастомных текстов
// =============================================================

// ID строк в FMG, которые мы используем под свои хинты
static constexpr int kCustomSmallHintTextId = 15113006;
static constexpr int kCustomBigHintHeaderId = 15113014;

// Максимальная длина хранимого кастомного текста
static constexpr size_t kMaxHintTextLen = 256;

// Глобальные буферы для текста, который прилетает из C#
static wchar_t g_CustomSmallHintText[kMaxHintTextLen] = L"";
static wchar_t g_CustomBigHintHeader[kMaxHintTextLen] = L"";

// =============================================================
// Внутренние типы и функции игры
// =============================================================

struct HintParamsA
{
    int hintId;        // 0x00
    int flags;         // 0x04
    int headerTextId;  // 0x08
    int bodyTextId;    // 0x0C
};

// Адреса внутренних функций / указателей в sekiro.exe
// (оставлены те, которые у тебя уже работали)
constexpr uintptr_t HINT_BASEPTR_RVA = 0x00E15E80;  // ShowHintBox_Internal
constexpr uintptr_t SMALL_HINT_FUNC_RVA = 0x00E15EE0;  // ShowSmallHintBoxSimple
constexpr uintptr_t REMOVE_HINTBOX_RVA = 0x00E131F0;  // RemoveHintBox_Internal

// Типы внутренних функций
using ShowSmallHintBoxSimple_t = void(__fastcall*)(HintParamsA* params, int hintId, int textId);
using ShowHintBox_Internal_t = void(__fastcall*)(HintParamsA* paramsA, void* paramsB);
using RemoveHintBox_Internal_t = void(__fastcall*)(int* hintId);

// Глобальные указатели на внутренние функции игры
static ShowSmallHintBoxSimple_t g_ShowSmallHintBoxSimple = nullptr;
static ShowHintBox_Internal_t   g_ShowHintBox_Internal = nullptr;
static RemoveHintBox_Internal_t g_RemoveHintBox_Internal = nullptr;

// RVA виртуальной функции, полученной через Lua:
// [HintVFuncLua] func RVA = 0xE2DEA0
static constexpr uintptr_t kHintTextVFuncRva = 0x00E2DEA0;

using HintTextVFunc_t = __int64(__fastcall*)(void* a1, void* a2, void* a3, void* a4);

static HintTextVFunc_t g_HintTextVFunc_Orig = nullptr;

// Вспомогательный лог, чтобы не спамить каждый кадр, если что-то пойдет не так
static void LogHintIdsOnce(uint32_t smallId, uint32_t headerId, uint32_t bodyId)
{
    static uint32_t lastSmall = 0xFFFFFFFF;
    static uint32_t lastHeader = 0xFFFFFFFF;
    static uint32_t lastBody = 0xFFFFFFFF;

    if (smallId == lastSmall && headerId == lastHeader && bodyId == lastBody)
        return;

    lastSmall = smallId;
    lastHeader = headerId;
    lastBody = bodyId;

    Logf("[HintTextVFunc] ids: small=%u header=%u body=%u",
        smallId, headerId, bodyId);
}

// Очень осторожный хук: сначала не трогаем память вообще, только лог указателей.
__int64 __fastcall Hook_HintTextVFunc(void* a1, void* a2, void* a3, void* a4)
{
    // сначала вызываем оригинал, чтобы не менять поведение
    __int64 result = 0;
    if (g_HintTextVFunc_Orig)
        result = g_HintTextVFunc_Orig(a1, a2, a3, a4);

    // Минимальная отладка: лог только адресов, без deref
    // (потом включим чтение, когда убедимся, что крэшей нет)
    Logf("[HintTextVFunc] a1=%p a2=%p a3=%p a4=%p", a1, a2, a3, a4);

    return result;
}

void InitHintEntryHook()
{
    auto base = (uintptr_t)GetModuleHandleW(L"sekiro.exe");
    LPVOID target = reinterpret_cast<LPVOID>(base + kHintTextVFuncRva);
    Logf("[InGameMessaging] HintTextVFunc target = %p (RVA=0x%zX)", target, kHintTextVFuncRva);

    MH_STATUS st = MH_CreateHook(
        target,
        &Hook_HintTextVFunc,
        reinterpret_cast<LPVOID*>(&g_HintTextVFunc_Orig)
    );
    if (st != MH_OK)
    {
        Logf("[InGameMessaging] MH_CreateHook(HintTextVFunc) failed: %d", (int)st);
    }
    st = MH_EnableHook(target);
    if (st != MH_OK)
    {
        Logf("[InGameMessaging] MH_EnableHook(HintTextVFunc) failed: %d", (int)st);
    }
}

// =============================================================
// Хук на TextMan-функцию, которая собирает текст (sekiro.exe+C7BCB0)
// =============================================================

using HintTextFunc_t = void* (__fastcall*)(void* a1, wchar_t* text, void* a3, unsigned int flags);
static HintTextFunc_t g_HintTextFunc_Orig = nullptr;

void* __fastcall Hook_HintTextFunc(void* a1, wchar_t* text, void* a3, unsigned int flags)
{
    // Сначала даём игре сделать всё как обычно
    void* result = g_HintTextFunc_Orig(a1, text, a3, flags);

    return result;
}

// =============================================================
// Авто-скрытие хинта через 5 секунд
// =============================================================

static volatile LONG g_LastHintTicket = 0;

DWORD WINAPI HintAutoHideThread(LPVOID param)
{
    LONG myTicket = static_cast<LONG>(reinterpret_cast<intptr_t>(param));

    Sleep(5000);

    if (myTicket != g_LastHintTicket)
        return 0;

    // Скрываем наш клиентский хинт (id=0)
    if (g_RemoveHintBox_Internal)
    {
        int id = 0;
        g_RemoveHintBox_Internal(&id);
    }

    return 0;
}

static void StartHintAutoHideTimer()
{
    LONG ticket = InterlockedIncrement(&g_LastHintTicket);

    HANDLE hThread = CreateThread(
        nullptr,
        0,
        HintAutoHideThread,
        reinterpret_cast<LPVOID>(static_cast<intptr_t>(ticket)),
        0,
        nullptr
    );

    if (hThread)
        CloseHandle(hThread);
}

// =============================================================
// Вспомогательные функции установки кастомного текста
// =============================================================

static void InGameMessaging_SetNextCustomSmallHintText(const wchar_t* text)
{
    if (!text)
        return;

    wcsncpy_s(g_CustomSmallHintText, text, _TRUNCATE);
}

static void InGameMessaging_SetNextCustomBigHeader(const wchar_t* header)
{
    if (!header)
        return;

    wcsncpy_s(g_CustomBigHintHeader, header, _TRUNCATE);
}

// =============================================================
// Публичный API (то, чем ты пользуешься из других частей DLL)
// =============================================================

void InGameMessaging_ShowSmallHintBoxSimple(int textId)
{
    if (!g_ShowSmallHintBoxSimple)
        return;

    int hintId = 0;
    HintParamsA params{};
    params.hintId = hintId;
    params.flags = 0;
    params.headerTextId = 0;
    params.bodyTextId = 0;

    g_ShowSmallHintBoxSimple(&params, textId, 0);
    StartHintAutoHideTimer();
}

void InGameMessaging_ShowSmallHintBox_ById(int textId)
{
    // Для простоты используем тот же простой вариант
    InGameMessaging_ShowSmallHintBoxSimple(textId);
}

void InGameMessaging_ShowHintBox_ById(int headerId, int textId)
{
    if (!g_ShowHintBox_Internal)
    {
        Log("[InGameMessaging] ShowHintBox_ById: internal function not ready");
        return;
    }

    HintParamsA params{};
    params.hintId = 0;
    params.flags = 0;
    params.headerTextId = headerId;
    params.bodyTextId = textId;

    // paramsB — это params + 0x08, как мы уже выяснили ранее
    void* paramsB = reinterpret_cast<void*>(
        reinterpret_cast<std::uint8_t*>(&params) + 0x08
        );

    Logf("[InGameMessaging] ShowHintBox_ById: headerId=%d, textId=%d", headerId, textId);

    g_ShowHintBox_Internal(&params, paramsB);

    StartHintAutoHideTimer();
}

void InGameMessaging_RemoveHintBox(int hintId)
{
    if (!g_RemoveHintBox_Internal)
    {
        Log("[InGameMessaging] RemoveHintBox: internal func not ready");
        return;
    }

    int id = hintId;

    Logf("[InGameMessaging] RemoveHintBox(%d)", id);
    g_RemoveHintBox_Internal(&id);
}

void InGameMessaging_RemoveClientHint()
{
    InGameMessaging_RemoveHintBox(0);
}

void InGameMessaging_ShowSmallHint(const wchar_t* text)
{
    if (!text)
        return;

    InGameMessaging_SetNextCustomSmallHintText(text);
    InGameMessaging_ShowSmallHintBoxSimple(kCustomSmallHintTextId);
}

void InGameMessaging_ShowHint(const wchar_t* header, const wchar_t* text)
{
    if (!text || !header)
        return;

    InGameMessaging_SetNextCustomBigHeader(header);
    InGameMessaging_SetNextCustomSmallHintText(text);
    InGameMessaging_ShowHintBox_ById(kCustomBigHintHeaderId, kCustomSmallHintTextId);
}

// =============================================================
// Помощники для пайпа (UTF-8 -> UTF-16)
// =============================================================

static std::wstring Utf8ToWide(const std::string& s)
{
    if (s.empty())
        return {};

    int len = MultiByteToWideChar(CP_UTF8, 0, s.c_str(), (int)s.size(), nullptr, 0);
    if (len <= 0)
        return {};

    std::wstring out(len, L'\0');
    MultiByteToWideChar(CP_UTF8, 0, s.c_str(), (int)s.size(), &out[0], len);
    return out;
}

void InGameMessaging_HandlePipeMessage(const std::string& msg)
{
    if (JsonIsType(msg, "show_hint"))
    {
        std::string text;
        std::string header;
        if (!JsonFieldToString(msg, "header", header))
            return;
        if (!JsonFieldToString(msg, "text", text))
            return;

        std::wstring wideHeader = Utf8ToWide(header);
        std::wstring wideText = Utf8ToWide(text);
        if (wideHeader.empty() || wideText.empty())
            return;

        InGameMessaging_ShowHint(wideHeader.c_str(), wideText.c_str());
    }
    else if (JsonIsType(msg, "show_small_hint"))
    {
        std::string utf8;
        if (!JsonFieldToString(msg, "text", utf8))
            return;

        std::wstring wide = Utf8ToWide(utf8);
        if (wide.empty())
            return;

        InGameMessaging_ShowSmallHint(wide.c_str());
    }
}

// =============================================================
// Инициализация (хуки + адреса внутренних функций)
// =============================================================

bool InitInGameMessaging()
{
    static bool s_Initialized = false;
    if (s_Initialized)
    {
        Log("[InGameMessaging] already initialized");
        return true;
    }

    // --- MinHook init ---
    {
        MH_STATUS st = MH_Initialize();
        if (st != MH_OK && st != MH_ERROR_ALREADY_INITIALIZED)
        {
            Logf("[InGameMessaging] MH_Initialize failed: %d", (int)st);
            return false;
        }
    }

    HMODULE hMod = GetModuleHandleW(L"sekiro.exe");
    if (!hMod)
    {
        Log("[InGameMessaging] GetModuleHandleW(sekiro.exe) failed");
        return false;
    }

    uintptr_t base = reinterpret_cast<uintptr_t>(hMod);

    // Внутренние функции игры
    g_RemoveHintBox_Internal = reinterpret_cast<RemoveHintBox_Internal_t>(base + REMOVE_HINTBOX_RVA);
    Logf("[InGameMessaging] RemoveHintBox_Internal = %p", g_RemoveHintBox_Internal);

    g_ShowHintBox_Internal = reinterpret_cast<ShowHintBox_Internal_t>(base + HINT_BASEPTR_RVA);
    Logf("[InGameMessaging] ShowHintBox_Internal = %p", g_ShowHintBox_Internal);

    g_ShowSmallHintBoxSimple = reinterpret_cast<ShowSmallHintBoxSimple_t>(base + SMALL_HINT_FUNC_RVA);
    Logf("[InGameMessaging] ShowSmallHintBoxSimple = %p", g_ShowSmallHintBoxSimple);

    // Хук на TextMan функцию (C7BCB0)
    void* hintTextTarget = reinterpret_cast<void*>(base + 0xC7BCB0);
    if (MH_CreateHook(hintTextTarget, &Hook_HintTextFunc,
        reinterpret_cast<void**>(&g_HintTextFunc_Orig)) != MH_OK)
    {
        Log("[InGameMessaging] MH_CreateHook failed for HintTextFunc");
        return false;
    }

    if (MH_EnableHook(hintTextTarget) != MH_OK)
    {
        Log("[InGameMessaging] MH_EnableHook failed for HintTextFunc");
        return false;
    }

    Logf("[InGameMessaging] Hook_HintTextFunc installed at %p", hintTextTarget);

    InitHintEntryHook();

    s_Initialized = true;
    Log("[InGameMessaging] InitInGameMessaging succeeded");
    return true;
}
